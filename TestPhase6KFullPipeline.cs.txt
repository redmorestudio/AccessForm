using System;
using System.IO;
using System.Threading.Tasks;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Logging;
using WordToPdfConverter.Models.Remediation;
using WordToPdfConverter.Services.Analysis;
using WordToPdfConverter.Services.Layout;
using WordToPdfConverter.Services.Pdf;
using WordToPdfConverter.Services.Phase6H;
using WordToPdfConverter.Services.Remediation;
using WordToPdfConverter.Services.Remediation.Models;
using WordToPdfConverter.Services.Remediation.Structure;

/// <summary>
/// Phase 6K Full Pipeline Test - Runs Erie PDF through complete remediation
/// including AI layout analysis, structure rebuild, and MCID linking
/// </summary>
public class TestPhase6KFullPipeline
{
    public static async Task Main(string[] args)
    {
        Console.WriteLine("================================================================================");
        Console.WriteLine("PHASE 6K FULL PIPELINE TEST - Erie Route 5 PDF");
        Console.WriteLine("================================================================================\n");

        // Input PDF
        var inputPdf = "/Users/sethredmore/Documents/Redmore Studio/AccessForm/WordToPdfConverter/StateAssets_archived_20251112_063442/Pennsylvania/Erie/ERA_0042_Route 229 August 2025.pdf";
        var outputPdf = "./erie_route229_phase6k_output.pdf";

        if (!File.Exists(inputPdf))
        {
            Console.WriteLine($"❌ Input PDF not found: {inputPdf}");
            return;
        }

        Console.WriteLine($"📄 Input: {inputPdf}");
        Console.WriteLine($"📤 Output: {outputPdf}\n");

        // Setup DI container
        var services = new ServiceCollection();

        // Logging
        services.AddLogging(builder =>
        {
            builder.AddConsole();
            builder.SetMinimumLevel(LogLevel.Information);
        });

        // Configuration
        var configuration = new ConfigurationBuilder()
            .AddInMemoryCollection(new[]
            {
                new KeyValuePair<string, string>("Phase6H:MicroserviceUrl", "http://localhost:8000"),
                new KeyValuePair<string, string>("AnthropicApiKey", "your-key-here") // Not used in this test
            })
            .Build();
        services.AddSingleton<IConfiguration>(configuration);

        // Job context with MCID enabled
        var jobContext = new RemediationJobContext
        {
            Options = new RemediationOptions
            {
                EnableMcidLinking = true,
                EnableMcidContentRewrite = true
            }
        };
        services.AddSingleton(jobContext);

        // Register services
        services.AddScoped<ILogicalLayoutAnalysisService, StubLogicalLayoutAnalysisService>();
        services.AddScoped<FormFieldEnrichmentService>();
        services.AddScoped<StructureTreeCleaner>();
        services.AddScoped<IPdfStructureWriter, ITextPdfStructureWriter>();
        services.AddScoped<SimpleTopDownLayoutEngine>();
        services.AddScoped<McidRewritePlanBuilder>();
        services.AddHttpClient<ExternalMcidRewriterService>();
        services.AddScoped<StructureRebuildService>();

        // Build provider
        var provider = services.BuildServiceProvider();
        using var scope = provider.CreateScope();

        var loggerFactory = scope.ServiceProvider.GetRequiredService<ILoggerFactory>();
        var logger = loggerFactory.CreateLogger<TestPhase6KFullPipeline>();

        // Get StructureRebuildService from DI
        var structureRebuildService = scope.ServiceProvider.GetRequiredService<StructureRebuildService>();

        // Load PDF
        var pdfBytes = await File.ReadAllBytesAsync(inputPdf);
        logger.LogInformation($"[TEST] Loaded {pdfBytes.Length / 1024}KB PDF with {CountPages(pdfBytes)} pages");

        // Run full structure rebuild (includes AI analysis + Phase 6K)
        logger.LogInformation("[TEST] ========== STARTING FULL STRUCTURE REBUILD ==========");

        try
        {
            var result = await structureRebuildService.RemediateAsync(pdfBytes);

            if (result.Success)
            {
                logger.LogInformation("[TEST] ✅ Structure rebuild successful!");
                logger.LogInformation($"[TEST] Duration: {result.Duration.TotalSeconds:F1}s");

                // Save output
                await File.WriteAllBytesAsync(outputPdf, result.OutputPdf);
                logger.LogInformation($"[TEST] Saved to: {outputPdf}");

                // Verify markers
                var bdcCount = CountOccurrences(result.OutputPdf, "BDC");
                var emcCount = CountOccurrences(result.OutputPdf, "EMC");

                logger.LogInformation($"[TEST] BDC markers: {bdcCount}");
                logger.LogInformation($"[TEST] EMC markers: {emcCount}");

                if (bdcCount > 0 && emcCount > 0)
                {
                    logger.LogInformation("[TEST] ✅✅✅ SUCCESS! Phase 6K full pipeline is working!");
                    logger.LogInformation("[TEST] - AI layout analysis complete");
                    logger.LogInformation("[TEST] - Structure tree built with MCR kids");
                    logger.LogInformation("[TEST] - BDC/EMC markers added to content streams");
                }
                else
                {
                    logger.LogWarning("[TEST] ⚠️  No markers found - may need investigation");
                }
            }
            else
            {
                logger.LogError($"[TEST] ❌ Structure rebuild failed: {result.ErrorMessage}");
            }
        }
        catch (Exception ex)
        {
            logger.LogError(ex, "[TEST] ❌ Exception during remediation");
        }

        Console.WriteLine("\n================================================================================");
        Console.WriteLine("Test complete!");
        Console.WriteLine("================================================================================");
    }

    private static int CountPages(byte[] pdfBytes)
    {
        try
        {
            using var stream = new MemoryStream(pdfBytes);
            using var pdf = new Syncfusion.Pdf.Parsing.PdfLoadedDocument(stream);
            return pdf.Pages.Count;
        }
        catch
        {
            return 0;
        }
    }

    private static int CountOccurrences(byte[] data, string search)
    {
        var text = System.Text.Encoding.Latin1.GetString(data);
        return text.Split(new[] { search }, StringSplitOptions.None).Length - 1;
    }
}
